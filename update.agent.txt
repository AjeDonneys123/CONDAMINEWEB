const fs = require('fs');
const path = require('path');

/**
 * 🛠️ APPLY.JS V11.0 - MODE ACTIF DIRECT
 * - Écrase les fichiers dès réception du tag [[[£ FILE: ... £]]]
 * - Utilisé quand l'Agent Antigravity est désactivé.
 */

const inputFile = 'update.txt';
const statusFile = 'apply_status.json';

function writeStatus(type, message) {
    const data = { status: type, message, timestamp: Date.now() };
    try { fs.writeFileSync(statusFile, JSON.stringify(data, null, 2)); } catch(e) {}
}

function applyUpdates() {
    try {
        if (!fs.existsSync(inputFile)) return;
        const content = fs.readFileSync(inputFile, 'utf8');
        if (!content || content.length < 10) return;

        console.log("🚀 [APPLY] Traitement de la mise à jour...");
        const fileBlocks = content.split('[[[£ FILE:');
        fileBlocks.shift(); // Enlever l'intro

        for (const block of fileBlocks) {
            const [header, ...codeLines] = block.split(']]]\n');
            const filePath = header.trim();
            let code = codeLines.join(']]]\n').split('[[[£ END:')[0].trim();

            const fullPath = path.join(__dirname, filePath);
            const dir = path.dirname(fullPath);

            if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
            
            fs.writeFileSync(fullPath, code);
            console.log(`✅ [UPDATED] ${filePath}`);
        }

        fs.writeFileSync(inputFile, ''); // Vider l'input
        writeStatus('OK', 'Mise à jour appliquée avec succès');
        console.log("✨ [APPLY] Terminé.");

    } catch (e) {
        console.log(`💥 [ERROR] ${e.message}`);
        writeStatus('ERROR', e.message);
    }
}

setInterval(applyUpdates, 1000);
console.log("⚡ APPLY V11.0 : MODE ACTIF (ÉCRASEMENT DIRECT) ACTIVÉ");
